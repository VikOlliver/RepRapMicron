# These are the GRBL config values used in the RAMPS PIKA V0.03
# Note: This is not a valid GCODE file.
# Also you cannot just poke this at GRBL with cncjs-app!
# Each instruction needs entering separately. Try using minicom with
# a 1 second inter-line delay or something.
$0=10 (Step pulse time, microseconds)
$1=255 (Step idle delay, milliseconds)
$2=4 (Step pulse invert, mask)
$3=3 (Step direction invert, mask)
$4=0 (Invert step enable pin, boolean)
$5=0 (Invert limit pins, boolean)
$6=0 (Invert probe pin, boolean)
$10=17 (Status report options, mask)
$11=0.100 (Junction deviation, millimeters)
$12=0.200 (Arc tolerance, millimeters)
$13=0 (Report in inches, boolean)
$20=0 (Soft limits enable, boolean)
$21=1 (Hard limits enable, boolean)
$22=1 (Homing cycle enable, boolean)
$23=3 (Homing direction invert, mask)
$24=5000.000 (Homing locate feed rate, mm/min)
$25=15000.000 (Homing search seek rate, mm/min)
$26=10 (Homing switch debounce delay, milliseconds)
$27=2000.000 (Homing switch pull-off distance, millimeters)
$30=1000 (Maximum spindle speed, RPM)
$31=0 (Minimum spindle speed, RPM)
$32=0 (Laser-mode enable, boolean)
$100=31.000 (X-axis travel resolution, step/mm)
$101=40.000 (Y-axis travel resolution, step/mm)
$102=32.000 (Z-axis travel resolution, step/mm)
$110=20000.000 (X-axis maximum rate, mm/min)
$111=20000.000 (Y-axis maximum rate, mm/min)
$112=20000.000 (Z-axis maximum rate, mm/min)
$120=200.000 (X-axis acceleration, mm/sec^2)
$121=200.000 (Y-axis acceleration, mm/sec^2)
$122=900.000 (Z-axis acceleration, mm/sec^2)
$130=7000.000 (X-axis maximum travel, millimeters)
$131=7000.000 (Y-axis maximum travel, millimeters)
$132=7000.000 (Z-axis maximum travel, millimeters)
